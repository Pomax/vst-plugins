# Dragging the pointer through the document selects what it crosses.
#
# The proof is that typing afterwards replaces the selection rather than being
# inserted at a caret: if nothing was selected, the text would still be there.
type:hello world

# From two characters in to past the end of the line. The press has to land
# inside the text: a line is only as wide as what is written on it, and a press
# beside it belongs to no line at all.
drag:24,155,90,155
type:X

expect:tabs is 1
expect:tab 0 is heX
