# The plugin's window fills the host's, at whatever size the host is.
#
# The plugin is a window of its own inside the host's, and the host used to put
# it right on the next frame of its own drawing loop. That is a frame late for
# every frame of a drag: the host paints its new shape while the plugin still
# has the old one, so the edge of the window shows the host's background where
# the plugin has not caught up. Now the plugin's window is moved from inside
# the resize itself, so no size the host is ever set to can leave the two
# disagreeing.
#
# The strip across the top is 26 points tall, so the plugin gets everything
# below that: the same width, and the height less 26.
type:something to keep

# Resizing is a hand on the corner, moving: nothing here sets a size any
# other way. Filmed from before the hand takes the corner, over a region big
# enough for the largest size the window is dragged to, since the region is
# fixed once recording starts: what the window does while it is happening can
# then be watched rather than guessed at.
remove:%CACHE%\host-resize-grown.txt
remove:%CACHE%\host-resize-shrunk.txt
film:%CACHE%\host-resize-drag|1000,760
dragto:880,600,1000
wait:800
geometry:%CACHE%\host-resize-grown.txt
# Smaller as well as bigger: a window that only ever grows can hide the case
# where the plugin is left overhanging the frame it is in.
dragto:620,420,1000
wait:800
geometry:%CACHE%\host-resize-shrunk.txt
endfilm:

# The sizes are named exactly, because a grab that lands on the wrong pixel
# and drags nothing leaves the window as it was, and a test that only asked
# whether the plugin still fits would pass on that.
expect:file at %CACHE%\host-resize-grown.txt contains host=880x600
expect:file at %CACHE%\host-resize-grown.txt contains editor=0,26,880x574
# And the plugin fills it: nothing above it but the strip, nothing beside or
# below it at all.
expect:file at %CACHE%\host-resize-grown.txt contains inset=0,26,0,0
expect:file at %CACHE%\host-resize-shrunk.txt contains host=620x420
expect:file at %CACHE%\host-resize-shrunk.txt contains editor=0,26,620x394
expect:file at %CACHE%\host-resize-shrunk.txt contains inset=0,26,0,0
# And the plugin is still alive and still taking keys after all of that.
expect:sections is 1
expect:section 0 is something to keep
