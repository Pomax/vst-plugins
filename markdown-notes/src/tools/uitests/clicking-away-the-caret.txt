# Clicking where there is no line takes the caret away, and typing then has
# nowhere to go.
#
# The empty space below the last line is document, but it is not a line: there
# is nothing there to put a caret on. Clicking a line puts it back, at the end
# of that line when the click was to the right of it.
type:first line
type:{ENTER}second line

# Well below the text, in the empty half of the window.
click:450,400
wait:500
type:this must not appear
wait:500
shot:%CACHE%\clicking-away-the-caret.png

# Beside the first line, past the end of what is written on it: the caret goes
# to the end of that line, and typing lands there.
click:600,151
wait:500
type:!
wait:500

expect:sections is 1
expect:section 0 is first line!\n\nsecond line
